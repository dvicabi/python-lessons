### 8.א קריאת קוד

```python
class SolarSystem:
    def __init__(self, capacity, battery):
        self.capacity = capacity
        self.battery = battery
        self.buy = 0
        self.sell = 0

    def period(self, net):
        self.battery += net
        if self.battery > self.capacity:
            self.sell += self.battery - self.capacity
            self.battery = self.capacity
        elif self.battery < 0:
            self.buy += -self.battery
            self.battery = 0

a = SolarSystem(6, 2)
b = SolarSystem(3, 0)
a.period(8)
b.period(-4)
a.period(-3)
print(a.battery, a.buy, a.sell)
print(b.battery, b.buy, b.sell)
```
