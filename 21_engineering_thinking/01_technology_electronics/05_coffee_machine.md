### 8.א קריאת קוד

```python
class CoffeeMachine:
    def __init__(self, coffee, water):
        self.coffee = coffee
        self.water = water
        self.revenue = 0

    def espresso(self, paid):
        if paid >= 8 and self.coffee >= 1 and self.water >= 2:
            self.coffee -= 1
            self.water -= 2
            self.revenue += 8
            return paid - 8
        return paid

a = CoffeeMachine(2, 4)
b = CoffeeMachine(1, 1)
print(a.espresso(10))
print(b.espresso(10))
print(a.revenue, b.revenue)
```
