### 8.א קריאת קוד

```python
class Printer:
    def __init__(self, stock):
        self.stock = stock
        self.printed = 0

    def layer(self, thickness):
        if thickness > 0 and self.stock - thickness - 1 >= 2:
            self.stock -= thickness + 1
            self.printed += 1

a = Printer(10)
b = Printer(6)
a.layer(3)
b.layer(4)
a.layer(3)
print(a.stock, a.printed)
print(b.stock, b.printed)
```
