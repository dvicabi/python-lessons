### 7.א קריאת קוד

```python
def charge(left, right, case, budget):
    while budget > 0 and case > 0 and (left < 10 or right < 10):
        if left <= right:
            left += 1
        else:
            right += 1
        case -= 1
        budget -= 1
    return [left, right, case]

print(state)
print(charge(state[0], state[1], state[2], 2))
```

### 8.א קריאת קוד

```python
class Earbuds:
    def __init__(self, left, right, case):
        self.left = left
        self.right = right
        self.case = case

    def play(self, amount):
        if amount >= 0 and self.left >= amount and self.right >= amount:
            self.left -= amount
            self.right -= amount

a = Earbuds(6, 4, 8)
b = Earbuds(2, 2, 5)
a.play(3)
a.play(2)
b.play(1)
print(a.left, a.right, a.case)
print(b.left, b.right, b.case)
```
