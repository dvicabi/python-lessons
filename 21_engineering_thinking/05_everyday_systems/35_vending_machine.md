# מכונת חטיפים: מה קורה אם המוצר לא יוצא או שאין עודף?

מכונת חטיפים מדומה מקבלת בקשות, מלאי וכסף.

לקוח יכול לשלם מספיק ועדיין לא לקבל מוצר, אם אין עודף מדויק או אם מנגנון המסירה מדווח על תקלה.

תפקידכם להפריד בין בדיקה, מסירה ועדכון העסקה.

## כללי הסימולציה

מחירים שלמים ודמיוניים.

מטבעות עודף בערכים 1,2,5, בכמויות מוגבלות.

במודל הכסף שהוכנס נשמר בנפרד ממלאי העודף, ואינו זמין לחישוב העודף באותה עסקה.

כסף שהוכנס הוא סכום מדומה, לא פירוט מטבעות.

עסקה חוקית דורשת מוצר במלאי, תשלום מספיק, הרכב עודף מדויק, ודגל מסירה הצליח=1.

אם אחד התנאים נכשל, מחזירים תשלום תקין במלואו בלי לעדכן מלאי מוצר, קופת עודף או הכנסות.

העודף ממוזער לפי מספר מטבעות;

בשוויון הראשון בסריקה.

עודף אפס הוא הרכב תקין של אפס מטבעות.

**מטרת המשימה:** לבנות מערכת שלמה בהדרגה, לא רק להגיע לפלט נכון.

אפשר להתחיל ברמה המתאימה לחומר שנלמד.

אין צורך להכיר את הרמות הבאות.

בכל רמה יש חמש משימות משנה: שני מעקבים, שאלות ביקורת, כתיבה עצמאית וניסוי הנדסי.

אין להריץ את הקוד לפני מעקב ידני.

בחרו בעצמכם כיצד לבנות טבלת מעקב.

לא תינתן תבנית.

for a in range(coin1 + 1):
    for b in range(coin2 + 1):
        for c in range(coin5 + 1):
            tested += 1
            value = a + b * 2 + c * 5
            if value == due:
                matches += 1
                count = a + b + c
                print(a, b, c, count)
                if best == -1 or count < best:
                    best = count
                    best1 = a
                    best2 = b
                    best5 = c
if best == -1:
    print("NO CHANGE")

    print("Best:", best1, best2, best5, best)

    best = []

def quote(prices, stock, coins, item, paid):
    if item < 0 or item >= len(prices) or paid < 0:
        return ["INVALID", []]
    if stock[item] == 0 or paid < prices[item]:
        return ["DENIED", []]
    notes = change(coins, paid - prices[item])
    if len(notes) == 0:
        return ["NO CHANGE", []]
    return ["READY", notes]

class CoinBox:
    def __init__(self, coins):
        self.coins = coins

    def value(self):
        return self.coins[0] + self.coins[1] * 2 + self.coins[2] * 5

    def plan(self, due):
        best = []
        best_count = -1
        for a in range(self.coins[0] + 1):
            for b in range(self.coins[1] + 1):
                for c in range(self.coins[2] + 1):
                    if a + b * 2 + c * 5 == due:
                        count = a + b + c
                        if best_count == -1 or count < best_count:
                            best = [a, b, c]
                            best_count = count
        return best

    def take(self, notes):
        for i in range(3):
            self.coins[i] -= notes[i]

class Vending:
    def __init__(self, prices, stock, box):
        self.prices = prices
        self.stock = stock
        self.box = box
        self.revenue = 0
        self.refunds = 0
        self.changes = 0
        self.paid = 0
        self.history = []
        self.initial = box.value()

    def buy(self, item, paid, dispensed):
        status = "DENIED"
        refund = paid
        if item >= 0 and item < len(self.prices) and paid >= 0:
            due = paid - self.prices[item]
            if due >= 0 and self.stock[item] > 0:
                notes = self.box.plan(due)
                if len(notes) == 0:
                    status = "NO CHANGE"
                elif dispensed != 1:
                    status = "JAM"
                else:
                    self.box.take(notes)
                    self.stock[item] -= 1
                    self.revenue += self.prices[item]
                    self.changes += due
                    refund = 0
                    status = "OK"
        if paid >= 0:
            self.paid += paid
            self.refunds += refund
        self.history.append([item, status])
        return status

    def audit(self):
        cash_ok = self.initial == self.box.value() + self.changes
        sales_ok = self.paid == self.revenue + self.changes + self.refunds
        return [cash_ok, sales_ok]

box = CoinBox([0, 3, 1])
machine = Vending([7, 5], [2, 2], box)
print(machine.buy(0, 13, 0))
print(machine.audit())
print(machine.buy(0, 13, 1))
print(machine.buy(1, 10, 1))
print(machine.buy(1, 7, 1))
print(machine.buy(0, 7, 1))
print(machine.stock, box.coins)
print(machine.revenue, machine.refunds, machine.changes)
print(machine.history)
print(machine.audit())
