# קופה בסופר: איך מבצעים עסקה עם מבצעים ומלאי מוגבל?

בקופה מדומה לקוח קונה מספר מוצרים.

מחיר, כמות, הנחות ומלאי חייבים להתאים.

בעסקה אחת יכולים להופיע כמה פריטים מאותו סוג.

המטרה היא לחשב מחיר נכון ולא לבצע חיוב חלקי כאשר הסל כולו אינו אפשרי.

## כללי הסימולציה

כל הכמויות והמחירים מספרים שלמים בדויים.

מוצר בעל מחיר חיובי;

כמות חיובית.

מבצע: אם כמות כוללת של מוצר בסל לפחות 3, מחסירים 2 שקלים לכל יחידה מאותו מוצר (מחירים לפחות 2).

תשלום מספיק ומלאי מספיק נדרשים לכל הסל.

ברמות הראשונות מדמים מוצר אחד או שורת קנייה אחת.

ברמות 6–8 סוכמים כמויות לפי סוג לפני בדיקת מלאי ומבצע.

עסקה שנדחתה אינה משנה מלאי או הכנסה.

**מטרת המשימה:** לבנות מערכת שלמה בהדרגה, לא רק להגיע לפלט נכון.

אפשר להתחיל ברמה המתאימה לחומר שנלמד.

אין צורך להכיר את הרמות הבאות.

בכל רמה יש חמש משימות משנה: שני מעקבים, שאלות ביקורת, כתיבה עצמאית וניסוי הנדסי.

אין להריץ את הקוד לפני מעקב ידני.

בחרו בעצמכם כיצד לבנות טבלת מעקב.

לא תינתן תבנית.

        status = "NO FUNDS"

```python
stock = 15
price = 10
revenue = 0
sold = 0
discount_total = 0
ok = 0
denied = 0
invalid = 0
for i in range(7):
    quantity = int(input())
    paid = int(input())
    if quantity <= 0 or paid < 0:
        invalid += 1
        status = "INVALID"
    else:
        gross = price * quantity
        discount = 0
        if quantity >= 3:
            discount = 2 * quantity
        total = gross - discount
        if quantity > stock or paid < total:
            denied += 1
            status = "DENIED"
        else:
            stock -= quantity
            sold += quantity
            revenue += total
            discount_total += discount
            ok += 1
            status = "OK"
    print(i + 1, status, stock, revenue)
print(ok, denied, invalid)
print(revenue, discount_total, sold)
print("Check:", stock + sold)
```

def aggregate(cart, count):
    needed = []
    for i in range(count):
        needed.append(0)
    for row in cart:
        item = row[0]
        qty = row[1]
        if item < 0 or item >= count or qty <= 0:
            return []
        needed[item] += qty
    return needed

def quote(prices, stock, cart):
    needed = aggregate(cart, len(prices))
    if len(cart) == 0 or len(needed) == 0:
        return ["INVALID", 0, []]
    total = 0
    for i in range(len(prices)):
        if needed[i] > stock[i]:
            return ["NO STOCK", 0, needed]
        total += needed[i] * prices[i]
        if needed[i] >= 3:
            total -= needed[i] * 2
    return ["OK", total, needed]

def checkout(prices, stock, cart, paid):
    result = quote(prices, stock, cart)
    if result[0] != "OK":
        return result[0]
    if paid < result[1]:
        return "NO FUNDS"
    for i in range(len(stock)):
        stock[i] -= result[2][i]
    return paid - result[1]

class Cart:
    def __init__(self, count):
        self.needed = []
        for i in range(count):
            self.needed.append(0)
        self.units = 0

    def add(self, item, qty):
        if item < 0 or item >= len(self.needed) or qty <= 0:
            return False
        self.needed[item] += qty
        self.units += qty
        return True

class Inventory:
    def __init__(self, prices, stock):
        self.prices = prices
        self.stock = stock

    def quote(self, cart):
        if cart.units == 0 or len(cart.needed) != len(self.stock):
            return -1
        total = 0
        for i in range(len(self.stock)):
            qty = cart.needed[i]
            if qty > self.stock[i]:
                return -1
            total += qty * self.prices[i]
            if qty >= 3:
                total -= qty * 2
        return total

    def take(self, cart):
        if self.quote(cart) == -1:
            return False
        for i in range(len(self.stock)):
            self.stock[i] -= cart.needed[i]
        return True

    def report(self):
        result = []
        for qty in self.stock:
            result.append(qty)
        return result

class Checkout:
    def __init__(self, inventory):
        self.inventory = inventory
        self.revenue = 0
        self.completed = []
        self.history = []

    def pay(self, order, cart, paid):
        price = self.inventory.quote(cart)
        if order in self.completed:
            status = "DUPLICATE"
        elif price == -1:
            status = "DENIED"
        elif paid < price:
            status = "NO FUNDS"
        else:
            self.inventory.take(cart)
            self.revenue += price
            self.completed.append(order)
            status = "OK"
        self.history.append([order, status])
        return status

    def summary(self):
        return [len(self.completed), self.revenue, self.inventory.report()]
