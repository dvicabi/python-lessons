# כספומט: האם אפשר למשוך את הסכום המבוקש?

לקוח מבקש למשוך כסף ממכשיר מדומה.

המכשיר מחזיק מספר מוגבל של שטרות, ולכן יתרה גבוהה במכשיר אינה מבטיחה שאפשר להחזיר כל סכום.

נפתח תוכנה שמחפשת הרכב שטרות חוקי, ממזערת את מספר השטרות ומעדכנת יתרה רק לאחר אישור.

## כללי הסימולציה

כל הכספים, החשבונות והעסקאות הם דמה, ללא חיבור לבנק.

משתמשים בשטרות 20, 50, 100 ו-200.

סכום משיכה חיובי בלבד.

הרכב חוקי נותן בדיוק את הסכום ואינו חורג מהמלאי.

בוחרים מספר שטרות קטן ביותר;

בשוויון שומרים הרכב ראשון לפי סדר הסריקה.

ברמות 5–7 מתחילים בחיפוש מלא עם מלאי קטן לצורכי הוראה.

ברמה 8 חשבון חייב להחזיק יתרה מספקת, ומלאי וחשבון משתנים רק אם המשיכה מצליחה.

**מטרת המשימה:** לבנות מערכת שלמה בהדרגה, לא רק להגיע לפלט נכון.

אפשר להתחיל ברמה המתאימה לחומר שנלמד.

אין צורך להכיר את הרמות הבאות.

בכל רמה יש חמש משימות משנה: שני מעקבים, שאלות ביקורת, כתיבה עצמאית וניסוי הנדסי.

אין להריץ את הקוד לפני מעקב ידני.

בחרו בעצמכם כיצד לבנות טבלת מעקב.

לא תינתן תבנית.

### 5.א קריאת קוד

```python
stock50 = 2
best = -1
best20 = 0
best50 = 0
tested = 0
valid = 0
for n20 in range(stock20 + 1):
    for n50 in range(stock50 + 1):
        tested += 1
        value = n20 * 20 + n50 * 50
        if value == amount:
            valid += 1
            count = n20 + n50
            print("MATCH", n20, n50, count)
            if best == -1 or count < best:
                best = count
                best20 = n20
                best50 = n50
if best == -1:
    print("NO SOLUTION")
else:
    print("Best:", best20, best50, best)
print("Tested:", tested)
print("Matches:", valid)
```

```python
class Account:
    def __init__(self, balance):
        self.balance = balance
        self.withdrawn = 0

    def can_pay(self, amount):
        return amount > 0 and amount <= self.balance

    def debit(self, amount):
        if not self.can_pay(amount):
            return False
        self.balance -= amount
        self.withdrawn += amount
        return True

class ATM:
    def __init__(self, stock):
        self.stock = stock
        self.history = []
        self.paid = 0
        self.initial = self.cash()

    def cash(self):
        values = [20, 50, 100, 200]
        total = 0
        for i in range(4):
            total += self.stock[i] * values[i]
        return total

    def plan(self, amount):
        best = []
        best_count = -1
        for a in range(self.stock[0] + 1):
            for b in range(self.stock[1] + 1):
                for c in range(self.stock[2] + 1):
                    for d in range(self.stock[3] + 1):
                        value = a * 20 + b * 50 + c * 100 + d * 200
                        count = a + b + c + d
                        if value == amount:
                            if best_count == -1 or count < best_count:
                                best = [a, b, c, d]
                                best_count = count
        return best

    def withdraw(self, account, amount):
        if not account.can_pay(amount):
            status = "NO FUNDS"
        else:
            notes = self.plan(amount)
            if len(notes) == 0:
                status = "NO NOTES"
            else:
                account.debit(amount)
                for i in range(4):
                    self.stock[i] -= notes[i]
                self.paid += amount
                status = "OK"
        self.history.append([amount, status])
        return status

    def audit(self):
        return self.initial == self.cash() + self.paid

a = Account(200)
b = Account(100)
machine = ATM([4, 2, 1, 1])
print(machine.withdraw(a, 60))
print(machine.withdraw(b, 90))
print(machine.withdraw(a, 140))
print(machine.withdraw(b, 20))
print(a.balance, b.balance)
print(a.withdrawn, b.withdrawn)
print(machine.stock)
print(machine.history)
print(machine.audit())
print(machine.paid == a.withdrawn + b.withdrawn)
```
