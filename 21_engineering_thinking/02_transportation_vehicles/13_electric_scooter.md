running = True
index = 0
while index < len(kinds) and running:
    i = index
    index += 1
    length = lengths[i]
    if kinds[i] == 3:
        battery += length
        if battery > 30:
            battery = 30
    else:
        cost = length
        if kinds[i] == 2:
            cost *= 2
        if battery - cost < 2:
            running = False
        else:
            battery -= cost
    if running:
        completed.append(i + 1)
        history.append(battery)
print(completed)
print(history)
print("Remaining:", len(kinds) - len(completed))
```

    while index < len(kinds) and running:
        i = index
        index += 1
        next_battery = step(battery, kinds[i], lengths[i])
        if next_battery == -1:
            running = False
        else:
            battery = next_battery
            history.append(battery)
    return history
print(route(22, [1, 2, 3, 2], [4, 5, 6, 6]))
print(route(25, [3, 2, 1], [10, 8, 6]))
```

class Scooter:
    def __init__(self, battery, capacity):
        self.battery = battery
        self.capacity = capacity
        self.distance = 0
        self.history = []

    def ride(self, kind, length):
        if length <= 0 or kind < 1 or kind > 3 or self.battery < 2:
            return False
        next_battery = self.battery
        if kind == 3:
            next_battery += length
            if next_battery > self.capacity:
                next_battery = self.capacity
        else:
            cost = length
            if kind == 2:
                cost *= 2
            if next_battery - cost < 2:
                return False
            next_battery -= cost
        self.battery = next_battery
        self.distance += length
        self.history.append(self.battery)
        return True

a = Scooter(20, 30)
b = Scooter(8, 15)
print(a.ride(2, 5))
print(b.ride(2, 5))
print(a.ride(3, 8))
print(b.ride(3, 12))
print(a.battery, a.distance, a.history)
print(b.battery, b.distance, b.history)
