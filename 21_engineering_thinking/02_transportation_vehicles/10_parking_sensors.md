### 7.א קריאת קוד

```python
def classify(distance):
    if distance < 0 or distance > 200:
        return "ERROR"
    if distance <= 20:
        return "STOP"
    if distance <= 50:
        return "NEAR"
    return "CLEAR"

def report(readings):
    result = []
    stops = 0
    for d in readings:
        state = classify(d)
        result.append(state)
        if state == "STOP":
            stops += 1
    print("Stops:", stops)
    return result

a = report([50, 20, 201, 51, 0])
b = report([80, 70, 60])
print(a)
print(b)
```

```python
class Sensor:
    def __init__(self):
        self.locked = False
        self.clear_count = 0
        self.history = []

    def read(self, distance):
        if distance < 0 or distance > 200:
            self.clear_count = 0
            self.history.append("ERROR")
            return "ERROR"
        if distance <= 20:
            self.locked = True
            self.clear_count = 0
        elif distance > 50:
            self.clear_count += 1
            if self.clear_count >= 2:
                self.locked = False
        else:
            self.clear_count = 0
        if self.locked:
            self.history.append("STOP")
            return "STOP"
        if distance <= 50:
            self.history.append("NEAR")
            return "NEAR"
        self.history.append("CLEAR")
        return "CLEAR"

a = Sensor()
b = Sensor()
print(a.read(15))
print(b.read(80))
print(a.read(60))
print(a.read(70))
print(b.read(-1))
print(a.history)
print(b.history)
```
