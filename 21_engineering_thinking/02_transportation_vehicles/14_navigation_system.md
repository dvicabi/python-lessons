## רמה 7

**פעולות (פונקציות) | קושי: מתקדמת+**

### 7.א קריאת קוד

```python
def total_times(bases, delays):
    times = []
    for i in range(len(bases)):
        times.append(bases[i] + delays[i])
    return times

def choose(times, prices, budget):
    best = -1
    for i in range(len(times)):
        if prices[i] > budget:
            continue
        if best == -1 or times[i] < times[best] or (times[i] == times[best] and prices[i] < prices[best]):
            best = i
    return best

times = total_times([18, 20, 24], [7, 3, 0])
prices = [4, 6, 0]
print(times)
print(choose(times, prices, 5))
print(choose(times, prices, 6))
print(choose(times, prices, 0))
```

```python
class Navigator:
    def __init__(self, budget):
        self.budget = budget
        self.selected = -1
        self.history = []

    def choose(self, times, prices):
        best = -1
        for i in range(len(times)):
            if prices[i] > self.budget:
                continue
            if best == -1 or times[i] < times[best] or (times[i] == times[best] and prices[i] < prices[best]):
                best = i
        self.selected = best
        self.history.append(best)
        return best

a = Navigator(5)
b = Navigator(8)
times = [25, 23, 24]
prices = [4, 6, 0]
print(a.choose(times, prices))
print(b.choose(times, prices))
times[2] += 5
print(a.choose(times, prices))
print(b.choose(times, prices))
print(a.history)
print(b.history)
```
