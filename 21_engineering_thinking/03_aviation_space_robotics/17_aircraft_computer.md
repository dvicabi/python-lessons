# מחשב בקרה במטוס: האם אפשר לסמוך על המדידה?

במעבדת סימולציה שני חיישנים דמיוניים מודדים אותו מדד.

## כללי הסימולציה

זהו מודל לימודי בדוי בלבד.

הפער הוא ההפרש החיובי בין המדידות.

רק כאשר שתיהן תקינות ומסכימות מחשבים ממוצע שלם;

ממוצע 70 ומעלה מציין REVIEW, אחרת NORMAL.

בשלבים מתקדמים נדרשות שלוש מדידות REVIEW רצופות כדי לסמן דוח לבדיקה.

INVALID או MISMATCH מאפסים רצף אך אינם מסירים סימון שכבר נשמר.

**מטרת המשימה:** לבנות מערכת שלמה בהדרגה, לא רק להגיע לפלט נכון.

## רמה 6

**רשימות בלבד | קושי: מתקדמת**

### 6.א קריאת קוד

```python
readings_a = [60, 72, 75, 69, 110, 73, 74, 75]
readings_b = [64, 76, 90, 71, 70, 75, 76, 77]
statuses = []
gaps = []
review_indexes = []
streak = 0
flag = False
for i in range(len(readings_a)):
    a = readings_a[i]
    b = readings_b[i]
    gap = b - a
    if gap < 0:
        gap = -gap
    gaps.append(gap)
    if a < 0 or a > 100 or b < 0 or b > 100:
        status = "INVALID"
    elif gap > 8:
        status = "MISMATCH"
    else:
        total = a + b
        average = total // 2
        if average >= 70:
            status = "REVIEW"
        else:
            status = "NORMAL"
    statuses.append(status)
    if status == "REVIEW":
        streak += 1
        review_indexes.append(i)
    else:
        streak = 0
    if streak >= 3:
        flag = True
    print(i, status, streak)
print(gaps)
print(review_indexes)
print(flag)
```

def classify(a, b, gap_limit, threshold):
    if a < 0 or a > 100 or b < 0 or b > 100:
        return "INVALID"
    gap = b - a
    if gap < 0:
        gap = -gap
    if gap > gap_limit:
        return "MISMATCH"
    total = a + b
    average = total // 2
    if average >= threshold:
        return "REVIEW"
    return "NORMAL"

def analyze(left, right, threshold):
    result = []
    streak = 0
    marks = []
    if len(left) == len(right):
        for i in range(len(left)):
            status = classify(left[i], right[i], 8, threshold)
            result.append(status)
            if status == "REVIEW":
                streak += 1
            else:
                streak = 0
            if streak == 3:
                marks.append(i)
    return [result, marks]

print(classify(70, 78, 8, 70))
print(classify(70, 79, 8, 70))
print(analyze([70, 72, 74, 60], [72, 74, 76, 62], 70))
print(analyze([70, 72], [72], 70))

class PairSensor:
    def __init__(self, gap_limit, threshold):
        self.gap_limit = gap_limit
        self.threshold = threshold

    def classify(self, a, b):
        if a < 0 or a > 100 or b < 0 or b > 100:
            return "INVALID"
        gap = b - a
        if gap < 0:
            gap = -gap
        if gap > self.gap_limit:
            return "MISMATCH"
        total = a + b
        average = total // 2
        if average >= self.threshold:
            return "REVIEW"
        return "NORMAL"

class Monitor:
    def __init__(self, sensor, required):
        self.sensor = sensor
        self.required = required
        self.streak = 0
        self.flag = False
        self.history = []

    def sample(self, a, b):
        status = self.sensor.classify(a, b)
        self.history.append(status)
        if status == "REVIEW":
            self.streak += 1
        else:
            self.streak = 0
        if self.streak >= self.required:
            self.flag = True
        return status

a = Monitor(PairSensor(8, 70), 2)
b = Monitor(PairSensor(4, 70), 2)
print(a.sample(70, 76))
print(b.sample(70, 76))
print(a.sample(72, 74))
print(b.sample(72, 74))
print(a.flag, a.streak, a.history)
print(b.flag, b.streak, b.history)
