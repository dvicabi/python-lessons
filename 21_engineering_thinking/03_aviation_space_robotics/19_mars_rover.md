# רובוט מחקר במאדים: כמה רחוק אפשר לחקור ועדיין לחזור?

כל המספרים בדויים;

בשלב הרשימות אוספים לכל היותר דגימה אחת מכל נקודה.

### 6.א קריאת קוד

```python
targets = [4, 8, 4, 10, 2, 0, 7]
energy = 32
position = 0
visited = []
sampled = []
denied = []
for i in range(len(targets)):
    target = targets[i]
    distance = target - position
    if distance < 0:
        distance = -distance
    if target < 0 or target > 12 or energy - distance < target:
        denied.append(i)
        print("DENIED", i)
    else:
        energy -= distance
        position = target
        visited.append(position)
        if position > 0 and position not in sampled and energy - 3 >= position:
            sampled.append(position)
            energy -= 3
        print(i, position, energy, sampled)
print(visited)
print(sampled)
print(denied)
print("Margin:", energy - position)
```

```python
def move(position, energy, target):
    distance = target - position
    if distance < 0:
        distance = -distance
    if target < 0 or target > 12 or energy - distance < target:
        return [position, energy, False]
    return [target, energy - distance, True]

def mission(targets, energy):
    position = 0
    sampled = []
    history = []
    for target in targets:
        result = move(position, energy, target)
        if result[2]:
            position = result[0]
            energy = result[1]
            if position > 0 and position not in sampled and energy - 3 >= position:
                sampled.append(position)
                energy -= 3
        history.append([position, energy])
    return [sampled, history]

print(move(4, 15, 9))
print(move(4, 14, 9))
print(mission([4, 8, 4, 10, 0], 32))
print(mission([4, 8, 4, 10, 0], 22))
```

### 8.א קריאת קוד

```python
class Rover:
    def __init__(self, energy):
        self.energy = energy
        self.position = 0
        self.sampled = []

    def move(self, target):
        distance = target - self.position
        if distance < 0:
            distance = -distance
        if target < 0 or target > 12 or self.energy - distance < target:
            return False
        self.position = target
        self.energy -= distance
        return True

    def collect(self):
        if self.position == 0 or self.position in self.sampled or self.energy - 3 < self.position:
            return False
        self.sampled.append(self.position)
        self.energy -= 3
        return True

    def margin(self):
        return self.energy - self.position

class Mission:
    def __init__(self, rover):
        self.rover = rover
        self.denied = 0
        self.history = []

    def visit(self, target):
        if self.rover.move(target):
            self.rover.collect()
        else:
            self.denied += 1
        self.history.append([self.rover.position, self.rover.energy])

a = Mission(Rover(30))
b = Mission(Rover(20))
for target in [4, 8, 4, 0]:
    a.visit(target)
    b.visit(target)
print(a.rover.sampled, a.history, a.denied)
print(b.rover.sampled, b.history, b.denied)
print(a.rover.margin(), b.rover.margin())
```
