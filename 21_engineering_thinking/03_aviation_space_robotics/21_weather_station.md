## רמה 6

**רשימות בלבד | קושי: מתקדמת**

מחשבים ממוצע נע של שלוש מדידות רצופות.

מדידה חסרה אינה מוחלפת באפס.

אין מילונים, סטים או טאפל.

אין def.

### 6.א קריאת קוד

```python
temps = [24, 30, 36, 39, -1, 33, 42, 45, 30]
averages = []
ends = []
invalid_windows = 0
for i in range(len(temps)):
    if i >= 2:
        total = 0
        valid = True
        for j in range(3):
            value = temps[i - 2 + j]
            if value < 0 or value > 50:
                valid = False
            else:
                total += value
        if valid:
            average = total // 3
            averages.append(average)
            ends.append(i)
            print(i, average)
        else:
            invalid_windows += 1
            print(i, "INVALID")
print(averages)
print(ends)
print("Invalid windows:", invalid_windows)
```

### 7.א קריאת קוד

def valid(value):
    return value >= 0 and value <= 50

def average_at(temps, end, size):
    if size <= 0 or end < size - 1 or end >= len(temps):
        return -1
    total = 0
    ok = True
    for j in range(size):
        value = temps[end - size + 1 + j]
        if valid(value):
            total += value
        else:
            ok = False
    if ok:
        return total // size
    return -1

def moving(temps, size):
    result = []
    for i in range(len(temps)):
        result.append(average_at(temps, i, size))
    return result

temps = [20, 28, 36, 50, -1, 30, 40]
print(average_at(temps, 3, 3))
print(moving(temps, 2))
print(moving(temps, 3))
print(moving(temps, 0))
```

class WeatherStation:
    def __init__(self, name):
        self.name = name
        self.temps = []
        self.valid_count = 0
        self.total = 0

    def record(self, temp):
        if temp < 0 or temp > 50:
            self.temps.append(-1)
        else:
            self.temps.append(temp)
            self.valid_count += 1
            self.total += temp

    def average(self):
        if self.valid_count == 0:
            return -1
        return self.total // self.valid_count

    def coverage(self):
        return [self.valid_count, len(self.temps)]

class Network:
    def __init__(self, stations):
        self.stations = stations

    def best_coverage(self):
        best = -1
        for i in range(len(self.stations)):
            data = self.stations[i].coverage()
            if data[1] > 0:
                if best == -1:
                    best = i
                else:
                    old = self.stations[best].coverage()
                    if data[0] * old[1] > old[0] * data[1]:
                        best = i
        return best

a = WeatherStation("A")
b = WeatherStation("B")
for value in [20, 40, -1, 30]:
    a.record(value)
for value in [30, 30, -1, -1, 40]:
    b.record(value)
network = Network([a, b])
print(a.average(), a.coverage())
print(b.average(), b.coverage())
print(network.best_coverage())
```
