## רמה 6

**רשימות בלבד | קושי: מתקדמת**

לכל הודעה מזהה.

תור FIFO שומר את סדר הקבלה, ומונה first מציין את ההודעה הבאה.

מותר רשימות, אינדקסים, len ו-append.

אין מילונים, סטים או טאפל.

אין def.

### 6.א קריאת קוד

```python
arrivals = [101, 102, 103, 104, 105, 106, 107, 108]
rates = [0, 0, 0, 0, 0, 2, 1, 5]
buffer = []
sent = []
dropped = []
first = 0
capacity = 5
for i in range(len(arrivals)):
    waiting = len(buffer) - first
    if waiting < capacity:
        buffer.append(arrivals[i])
    else:
        dropped.append(arrivals[i])
    count = 0
    while count < rates[i] and first < len(buffer):
        sent.append(buffer[first])
        first += 1
        count += 1
    print(i + 1, len(buffer) - first, count)
pending = []
for i in range(len(buffer)):
    if i >= first:
        pending.append(buffer[i])
print(sent)
print(dropped)
print(pending)
print("Check:", len(sent) + len(dropped) + len(pending))
```

def receive(buffer, first, message, capacity):
    if len(buffer) - first >= capacity:
        return False
    buffer.append(message)
    return True

def transmit(buffer, first, rate, sent):
    count = 0
    while count < rate and first < len(buffer):
        sent.append(buffer[first])
        first += 1
        count += 1
    return first

def simulate(arrivals, rates, capacity):
    buffer = []
    sent = []
    dropped = []
    first = 0
    if len(arrivals) == len(rates):
        for i in range(len(arrivals)):
            if not receive(buffer, first, arrivals[i], capacity):
                dropped.append(arrivals[i])
            first = transmit(buffer, first, rates[i], sent)
    return [sent, dropped, len(buffer) - first]

print(simulate([11, 12, 13, 14, 15], [0, 0, 0, 1, 4], 3))
print(simulate([11, 12, 13, 14, 15], [0, 0, 0, 1, 4], 2))
print(simulate([11], [], 3))
```

class Channel:
    def __init__(self, capacity):
        self.capacity = capacity
        self.buffer = []
        self.first = 0
        self.sent = []
        self.dropped = []

    def pending(self):
        return len(self.buffer) - self.first

    def receive(self, message):
        if self.pending() >= self.capacity:
            self.dropped.append(message)
            return False
        self.buffer.append(message)
        return True

    def transmit(self, rate):
        count = 0
        while count < rate and self.first < len(self.buffer):
            self.sent.append(self.buffer[self.first])
            self.first += 1
            count += 1
        return count

class Station:
    def __init__(self, left, right):
        self.left = left
        self.right = right
        self.windows = 0

    def tick(self, message, rate_left, rate_right):
        self.windows += 1
        self.left.receive(message)
        self.right.receive(message)
        print(self.windows, self.left.transmit(rate_left), self.right.transmit(rate_right))

a = Channel(2)
b = Channel(4)
station = Station(a, b)
station.tick(101, 0, 0)
station.tick(102, 0, 1)
station.tick(103, 1, 0)
station.tick(104, 3, 3)
print(a.sent, a.dropped, a.pending())
print(b.sent, b.dropped, b.pending())
```
