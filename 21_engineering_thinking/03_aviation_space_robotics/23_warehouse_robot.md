## רמה 6

**רשימות בלבד | קושי: מתקדמת**

משתמשים במפת רשימות דו-ממדית.

בתרחיש הקריאה כל הפקודות הן כיוונים חוקיים בין 1 ל-4. בתרגיל העצמאי תוסיפו גם בדיקת כיוון לא חוקי.

איסוף משנה את המשבצת רק לאחר תנועה חוקית.

### 6.א קריאת קוד

```python
grid = [[0, 2, 0, 0, 0], [0, 0, 1, 2, 0], [0, 2, 1, 0, 0], [0, 0, 2, 0, 2]]
commands = [2, 3, 2, 3, 3, 2, 2, 2, 4, 2]
row = 0
col = 0
collected = 0
history = []
picked = []
for direction in commands:
    nr = row
    nc = col
    if direction == 1:
        nr -= 1
    elif direction == 2:
        nc += 1
    elif direction == 3:
        nr += 1
    elif direction == 4:
        nc -= 1
    if nr >= 0 and nr < len(grid) and nc >= 0 and nc < len(grid[nr]):
        if grid[nr][nc] != 1:
            row = nr
            col = nc
            if grid[row][col] == 2:
                collected += 1
                grid[row][col] = 0
                picked.append([row, col])
    history.append([row, col])
print(history)
print(picked)
print(collected)
print(grid)
```

```python
def move(grid, row, col, direction):
    nr = row
    nc = col
    if direction == 1:
        nr -= 1
    elif direction == 2:
        nc += 1
    elif direction == 3:
        nr += 1
    elif direction == 4:
        nc -= 1
    else:
        return [row, col, "INVALID"]
    if nr < 0 or nr >= len(grid) or nc < 0 or nc >= len(grid[nr]):
        return [row, col, "OUTSIDE"]
    if grid[nr][nc] == 1:
        return [row, col, "BLOCKED"]
    return [nr, nc, "MOVED"]

def walk(grid, commands):
    row = 0
    col = 0
    statuses = []
    positions = []
    for direction in commands:
        result = move(grid, row, col, direction)
        row = result[0]
        col = result[1]
        statuses.append(result[2])
        positions.append([row, col])
    return [statuses, positions]

grid = [[0, 2, 0, 0, 0], [0, 0, 1, 2, 0], [0, 2, 1, 0, 0], [0, 0, 2, 0, 2]]
print(move(grid, 1, 1, 2))
print(move(grid, 0, 0, 1))
print(walk(grid, [2, 3, 2, 3, 3, 2]))
print(grid)
```

```python
class Warehouse:
    def __init__(self, grid):
        self.grid = grid

    def allowed(self, row, col):
        if row < 0 or row >= len(self.grid):
            return False
        if col < 0 or col >= len(self.grid[row]):
            return False
        return self.grid[row][col] != 1

    def pick(self, row, col):
        if self.grid[row][col] == 2:
            self.grid[row][col] = 0
            return True
        return False

class Robot:
    def __init__(self, warehouse):
        self.warehouse = warehouse
        self.row = 0
        self.col = 0
        self.picked = []
        self.denied = 0

    def move(self, direction):
        nr = self.row
        nc = self.col
        if direction == 1:
            nr -= 1
        elif direction == 2:
            nc += 1
        elif direction == 3:
            nr += 1
        elif direction == 4:
            nc -= 1
        else:
            self.denied += 1
            return False
        if not self.warehouse.allowed(nr, nc):
            self.denied += 1
            return False
        self.row = nr
        self.col = nc
        if self.warehouse.pick(nr, nc):
            self.picked.append([nr, nc])
        return True

warehouse = Warehouse([[0, 2, 0], [0, 0, 1], [0, 2, 0]])
a = Robot(warehouse)
b = Robot(warehouse)
print(a.move(2), b.move(2))
print(a.move(3), b.move(3))
print(a.move(3), b.move(2))
print(a.picked, b.picked)
print(a.denied, b.denied)
print(warehouse.grid)
```
